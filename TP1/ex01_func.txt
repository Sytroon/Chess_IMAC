#define GLFW_INCLUDE_NONE
#include <glad/glad.h>
#include <GLFW/glfw3.h>

#include <glimac/Sphere.hpp>
#include <glimac/glm.hpp>
#include <glimac/Program.hpp> 
#include <glimac/FilePath.hpp>
#include <glimac/TrackballCamera.hpp>

// =====================
// GLOBAL STATE
// =====================

static glimac::TrackballCamera camera;

static GLuint vao;
static GLuint vbo;
static glimac::Sphere* sphere = nullptr;

static glimac::Program program;
static GLuint uMVPMatrixLoc;
static GLuint uMVMatrixLoc;
static GLuint uNormalMatrixLoc;

static glm::mat4 ProjMatrix;

// =====================
// INIT
// =====================

void init3D(const char* executablePath)
{
    // Shader
    glimac::FilePath applicationPath(executablePath);
    program = loadProgram(
        applicationPath.dirPath() + "TP1/shaders/3D.vs.glsl",
        applicationPath.dirPath() + "TP1/shaders/normals.fs.glsl"
    );

    // if (program.getGLId() == 0) {
    //     std::cerr << "Shader loading failed!" << std::endl;
    // }

    program.use();

    uMVPMatrixLoc = glGetUniformLocation(program.getGLId(), "uMVPMatrix");
    uMVMatrixLoc = glGetUniformLocation(program.getGLId(), "uMVMatrix");
    uNormalMatrixLoc = glGetUniformLocation(program.getGLId(), "uNormalMatrix");

    // Sphere
    sphere = new glimac::Sphere(1, 32, 16);

    // VBO
    glGenBuffers(1, &vbo);
    glBindBuffer(GL_ARRAY_BUFFER, vbo);
    glBufferData(GL_ARRAY_BUFFER,
        sphere->getVertexCount() * sizeof(glimac::ShapeVertex),
        sphere->getDataPointer(),
        GL_STATIC_DRAW
    );
    glBindBuffer(GL_ARRAY_BUFFER, 0);

    // VAO
    glGenVertexArrays(1, &vao);
    glBindVertexArray(vao);
    glBindBuffer(GL_ARRAY_BUFFER, vbo);

    // position
    glEnableVertexAttribArray(0);
    glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE,
        sizeof(glimac::ShapeVertex),
        (void*)offsetof(glimac::ShapeVertex, position));

    // normal
    glEnableVertexAttribArray(1);
    glVertexAttribPointer(1, 3, GL_FLOAT, GL_FALSE,
        sizeof(glimac::ShapeVertex),
        (void*)offsetof(glimac::ShapeVertex, normal));

    // texCoords
    glEnableVertexAttribArray(2);
    glVertexAttribPointer(2, 2, GL_FLOAT, GL_FALSE,
        sizeof(glimac::ShapeVertex),
        (void*)offsetof(glimac::ShapeVertex, texCoords));

    glBindBuffer(GL_ARRAY_BUFFER, 0);
    glBindVertexArray(0);

    glEnable(GL_DEPTH_TEST);

    // Projection
    ProjMatrix = glm::perspective(
        glm::radians(70.f),
        1.0f, // ratio → sera amélioré plus tard avec ImGui
        0.1f,
        100.f
    );
}

// =====================
// RENDER
// =====================

void render3D(float time)
{
    program.use();

    glBindVertexArray(vao);

    // ===== PLANET =====
    glm::mat4 ViewMatrix = camera.getViewMatrix();

    glm::mat4 ModelMatrix = glm::mat4(1.0f);
    glm::mat4 MVMatrix = ViewMatrix * ModelMatrix;
    glm::mat4 NormalMatrix = glm::transpose(glm::inverse(MVMatrix));
    glm::mat4 MVP = ProjMatrix * MVMatrix;

    glUniformMatrix4fv(uMVPMatrixLoc, 1, GL_FALSE, glm::value_ptr(MVP));
    glUniformMatrix4fv(uMVMatrixLoc, 1, GL_FALSE, glm::value_ptr(MVMatrix));
    glUniformMatrix4fv(uNormalMatrixLoc, 1, GL_FALSE, glm::value_ptr(NormalMatrix));

    glDrawArrays(GL_TRIANGLES, 0, sphere->getVertexCount());

    // ===== MOON =====
    glm::mat4 ModelMatrixMoon = glm::mat4(1.0f);
    ModelMatrixMoon = glm::translate(ModelMatrixMoon, glm::vec3(0, 0, -2));
    ModelMatrixMoon = glm::rotate(ModelMatrixMoon, time, glm::vec3(0, 0, 1));
    ModelMatrixMoon = glm::translate(ModelMatrixMoon, glm::vec3(-2, 0, 0));
    ModelMatrixMoon = glm::scale(ModelMatrixMoon, glm::vec3(0.2, 0.2, 0.2));

    glm::mat4 MVMatrixMoon = ViewMatrix * ModelMatrixMoon;
    glm::mat4 NormalMatrixMoon = glm::transpose(glm::inverse(MVMatrixMoon));
    glm::mat4 MVPMoon = ProjMatrix * MVMatrixMoon;

    glUniformMatrix4fv(uMVPMatrixLoc, 1, GL_FALSE, glm::value_ptr(MVPMoon));
    glUniformMatrix4fv(uMVMatrixLoc, 1, GL_FALSE, glm::value_ptr(MVMatrixMoon));
    glUniformMatrix4fv(uNormalMatrixLoc, 1, GL_FALSE, glm::value_ptr(NormalMatrixMoon));

    glDrawArrays(GL_TRIANGLES, 0, sphere->getVertexCount());

    glBindVertexArray(0);
}

int main(int argc, char* argv[]) {

    std::cout << "Start program" << std::endl;

    if (!glfwInit()) {
        std::cerr << "GLFW init failed" << std::endl;
        return -1;
    }

    std::cout << "GLFW OK" << std::endl;

    GLFWwindow* window = glfwCreateWindow(800, 800, "Test 3D", nullptr, nullptr);
    if (!window) {
        std::cerr << "Window creation failed" << std::endl;
        glfwTerminate();
        return -1;
    }

    std::cout << "Window OK" << std::endl;

    glfwMakeContextCurrent(window);

    if (!gladLoadGLLoader((GLADloadproc)glfwGetProcAddress)) {
        std::cerr << "GLAD failed" << std::endl;
        return -1;
    }

    std::cout << "GLAD OK" << std::endl;

    init3D(argv[0]);

    while (!glfwWindowShouldClose(window)) {
        float time = glfwGetTime();

        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

        render3D(time);

        glfwSwapBuffers(window);
        glfwPollEvents();
    }

    glfwTerminate();
    return 0;
}
